Задание

Нужно вывести только те города, где средняя зарплата больше 100 000.


with tt as (   
    select  
    city,
    avg(salary) as t1
    from employees
    group by city
)

select 
*
from tt
where t1 > 100000;





Здесь мы впервые столкнём CTE с JOIN.
Используем CTE + обычную таблицу одновременно.

Нужно вывести имя сотрудника, его зарплату и среднюю зарплату его города:


with tt as (
    SELECT
    city,
    avg(salary) as avg_s
    from employees
    GROUP BY city

)


select 

t.name,
t.salary,
t1.avg_s
from employees t
join tt t1
on t.city = t1.city
where t.salary > t1.avg_s;
