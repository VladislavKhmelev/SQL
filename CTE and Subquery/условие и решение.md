Задание

Найди отделы, в которых:

средняя зарплата выше средней зарплаты по всей компании;
в отделе работает минимум 2 человека.


with tt as (
    select
    *
    from departments
    inner join employees
    on departments.id = employees.department_id
),

tt1 as (
    select 
    department,
    avg(salary) as ee,
    count(*) as ee1
    from tt
    group by department
)

select 
e.department,
from tt1 e
where e.ee > (
    select 
    avg(salary)
    from employees
)

and ee1 >= 2;
